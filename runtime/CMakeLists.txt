cmake_minimum_required(VERSION 3.16)
project(flightwall_data LANGUAGES CXX)
add_library(flightwall_data src/flight_data.cpp src/provider_policy.cpp src/community_config.cpp src/community_cache.cpp esp32/provider_wire.cpp esp32/stream_adsb.cpp esp32/http_body.cpp)
target_compile_features(flightwall_data PUBLIC cxx_std_17)
target_include_directories(flightwall_data PUBLIC src)
include(CTest)
if(BUILD_TESTING)
  foreach(test flight_data provider_policy provider_wire stream_adsb http_body community_cache)
    add_executable(${test}_test tests/${test}_test.cpp)
    target_link_libraries(${test}_test PRIVATE flightwall_data)
    add_test(NAME ${test} COMMAND ${test}_test)
  endforeach()
endif()
add_executable(feed_inspect tools/feed_inspect.cpp)
target_link_libraries(feed_inspect PRIVATE flightwall_data)
