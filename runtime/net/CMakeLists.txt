cmake_minimum_required(VERSION 3.16)
project(flightwall_source_tls LANGUAGES C CXX)
set(MBEDTLS_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../build/net-deps/mbedtls" CACHE PATH "Verified Mbed TLS source")
set(FW_TLS_TX_BYTES 2048 CACHE STRING "TLS transmit content buffer size")
option(FW_TLS_HOST_TOOLS "Build desktop probe and local tests" ON)
if(NOT EXISTS "${MBEDTLS_SOURCE}/.flightwall-source.json")
  message(FATAL_ERROR "Run net/fetch_source.py explicitly before configuring")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/fetch_source.py" --output "${MBEDTLS_SOURCE}"
                RESULT_VARIABLE verified OUTPUT_QUIET)
if(NOT verified EQUAL 0)
  message(FATAL_ERROR "Mbed TLS source verification failed")
endif()
if(NOT FW_TLS_TX_BYTES MATCHES "^(2048|16384)$")
  message(FATAL_ERROR "Only the tested 2048 and comparison 16384 profiles are supported")
endif()
file(GLOB tls_sources CONFIGURE_DEPENDS "${MBEDTLS_SOURCE}/library/*.c")
add_library(fw_mbedtls STATIC ${tls_sources})
target_include_directories(fw_mbedtls PUBLIC "${MBEDTLS_SOURCE}/include" "${CMAKE_CURRENT_SOURCE_DIR}"
                           PRIVATE "${MBEDTLS_SOURCE}/library")
target_compile_definitions(fw_mbedtls PUBLIC MBEDTLS_CONFIG_FILE="fw_mbedtls_config.h" FW_TLS_TX_BYTES=${FW_TLS_TX_BYTES})
if(WIN32)
  target_link_libraries(fw_mbedtls PUBLIC ws2_32)
endif()
add_library(fw_source_tls STATIC tls_client.cpp tls_port.cpp)
target_compile_features(fw_source_tls PUBLIC cxx_std_17)
target_link_libraries(fw_source_tls PRIVATE fw_mbedtls)
target_compile_options(fw_source_tls PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti)
target_include_directories(fw_source_tls PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_custom_command(TARGET fw_source_tls POST_BUILD
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/make_symbol_prefix.py"
          --nm "${CMAKE_NM}" --check $<TARGET_FILE:fw_source_tls> $<TARGET_FILE:fw_mbedtls>
  VERBATIM)
include(CTest)
if(FW_TLS_HOST_TOOLS)
  add_executable(tls_probe host_probe.cpp ../esp32/http_body.cpp)
  target_link_libraries(tls_probe PRIVATE fw_source_tls fw_mbedtls)
  # Same probe, but all socket/RNG/clock IO goes through the ESP32 adapter.
  add_executable(tls_probe_esp_transport host_probe.cpp esp32_transport.c ../esp32/http_body.cpp)
  target_compile_definitions(tls_probe_esp_transport PRIVATE FW_PROBE_ESP_TRANSPORT)
  target_compile_options(tls_probe_esp_transport PRIVATE $<$<COMPILE_LANGUAGE:C>:-Wall -Wextra -Werror -std=c11>)
  target_link_libraries(tls_probe_esp_transport PRIVATE fw_source_tls fw_mbedtls)
  if(WIN32)
    target_link_libraries(tls_probe PRIVATE ws2_32 bcrypt)
    target_link_libraries(tls_probe_esp_transport PRIVATE ws2_32 bcrypt)
  endif()
endif()
if(BUILD_TESTING AND FW_TLS_HOST_TOOLS)
  add_executable(tls_lifecycle_test tests/tls_lifecycle_test.cpp)
  target_link_libraries(tls_lifecycle_test PRIVATE fw_source_tls fw_mbedtls)
  target_compile_options(tls_lifecycle_test PRIVATE -UNDEBUG)
  add_test(NAME tls_lifecycle COMMAND tls_lifecycle_test)
  add_executable(tls_c_api_test tests/tls_c_api_test.c)
  target_link_libraries(tls_c_api_test PRIVATE fw_source_tls)
  target_compile_options(tls_c_api_test PRIVATE -UNDEBUG)
  add_test(NAME tls_c_api COMMAND tls_c_api_test)
  add_test(NAME tls_local_https COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_local_https.py" --probe $<TARGET_FILE:tls_probe>)
  add_executable(esp_transport_test tests/esp_transport_test.cpp esp32_transport.c)
  target_include_directories(esp_transport_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_options(esp_transport_test PRIVATE -UNDEBUG $<$<COMPILE_LANGUAGE:C>:-Wall -Wextra -Werror -std=c11>)
  if(WIN32)
    target_link_libraries(esp_transport_test PRIVATE ws2_32 bcrypt)
  endif()
  add_test(NAME esp_transport COMMAND esp_transport_test)
  add_test(NAME tls_local_https_esp_transport COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_local_https.py" --probe $<TARGET_FILE:tls_probe_esp_transport>)
endif()
